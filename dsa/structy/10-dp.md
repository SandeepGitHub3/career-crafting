

### max palin subsequence
Write a method, maxPalinSubsequence, that takes in a string as an argument. The method should return the length of the longest subsequence of the string that is also a palindrome.

A subsequence of a string can be created by deleting any characters of the string, while maintaining the relative order of characters.

Source.maxPalinSubsequence("luwxult"); // -> 5
Source.maxPalinSubsequence("lol"); // -> 3

```
public static int maxPalinSubsequence(String str) {
    return maxPalinSubsequence(str, 0, str.length()-1, new HashMap<>());
  }

  public static int maxPalinSubsequence(String str,int front, int back, Map<String,Integer> memo) {
    if(front>back) return 0;
    if(front == back) return 1;

    String key = front +"--"+ back;
    if(memo.containsKey(key)) return memo.get(key);
    
    int res = 0 ;
    if(str.charAt(front) == str.charAt(back)){
      res = 2 + maxPalinSubsequence(str, front+1, back -1,memo);
    }else{
      res = Math.max(
               maxPalinSubsequence(str, front+1, back,memo),
               maxPalinSubsequence(str, front, back-1,memo)
            );
    }
    memo.put(key,res);
    return res;
  }
```


#### overlap subsequence
Write a method, overlapSubsequence, that takes in two strings as arguments. The method should return the length of the longest overlapping subsequence.

A subsequence of a string can be created by deleting any characters of the string, while maintaining the relative order of characters.

Source.overlapSubsequence("dogs", "daogt"); // -> 3

```
public static int overlapSubsequence(String str1, String str2) {
    return overlapSubsequence(str1,str2,0,0,new HashMap<>());
  }

  public static int overlapSubsequence(String str1, String str2, int s1, int s2, Map<String,Integer> memo) {
    
    if(s1 >= str1.length() || s2 >= str2.length()) return 0;

    String key = s1 + "--" + s2;
    if(memo.containsKey(key)) return memo.get(key);
    
    int ovSRes = 0;
    
    if (str1.charAt(s1) == str2.charAt(s2))
      ovSRes = 1 + overlapSubsequence(str1,str2,s1+1,s2+1,memo);
    else
      ovSRes = Math.max(
        overlapSubsequence(str1,str2,s1+1,s2,memo),
        overlapSubsequence(str1,str2,s1,s2+1,memo)
      );

    memo.put(key, ovSRes);
    return ovSRes;
  }
```

```
public static boolean canConcat(String s, List<String> words) {
    return canConcat(s, words, 0, new HashMap<>());
  }
  
  public static boolean canConcat(String s, List<String> words, int i, HashMap<Integer, Boolean> memo) {
    if (i == s.length()) {
      return true;
    }
    
    if (memo.containsKey(i)) {
      return memo.get(i);
    }
    
    for (String word : words) {
      if (s.startsWith(word, i)) {
        if (canConcat(s, words, i + word.length(), memo)) {
          memo.put(i, true);
          return true;
        }
      }
    }
    
    memo.put(i, false);
    return false;
  }
```

### quickest concat
Write a method, quickestConcat, that takes in a string and a list of words as arguments. The method should return the minimum number of words needed to build the string by concatenating words of the list. If it is not possible to build the string, then return -1.

You may use words of the list as many times as needed.

```
public static int quickestConcat(String s, List<String> words) {
    return quickestConcat(s,words,0, new HashMap<>());
  }

  public static int quickestConcat(String s, List<String> words, int s1,Map<Integer,Integer> memo) {
    if(s1 >= s.length()) return 0;

    if(memo.containsKey(s1)) return memo.get(s1);
    
    int shortestConcat = Integer.MAX_VALUE;
    for(String word: words){
      if(s.substring(s1).startsWith(word)){
        int currRes = quickestConcat(s,words,s1+word.length(),memo);
        if(currRes >= 0){
          shortestConcat = Math.min(shortestConcat,currRes);
        }
      }
    }

    shortestConcat = shortestConcat == Integer.MAX_VALUE? -1:shortestConcat+1;
    memo.put(s1,shortestConcat);
    return shortestConcat;
  }
```

```
public static boolean validCompound(String compound, List<String> elements) {
    return validCompound(compound,elements,0, new HashMap<>());
  }

  private static boolean validCompound(String compound, List<String> elements,Integer c, Map<Integer, Boolean> memo) {
    if(c == compound.length()) return true;

    if(memo.containsKey(c)) return memo.get(c);
    boolean isValidCompound = false;

    for(String element: elements){
      if(compound.substring(c).startsWith(element.toLowerCase()) && validCompound(compound,elements,c+element.length(),memo)){
        isValidCompound = true;
        break;
      }
    }
    memo.put(c,isValidCompound);
    return isValidCompound;
  }
```

```
    return countCompounds(compound,elements,0, new HashMap<>());
  }

  private static int countCompounds(String compound, List<String> elements,Integer c, Map<Integer, Integer> memo) {
    if(c == compound.length()) return 1;

    if(memo.containsKey(c)) return memo.get(c);
    int compoundCount = 0;

    for(String element: elements){
      if(compound.substring(c).startsWith(element.toLowerCase())){
        int currCount = countCompounds(compound,elements,c+element.length(),memo);
        compoundCount = compoundCount + currCount;
      } 
    }
    memo.put(c,compoundCount);
    return compoundCount;
  }
```

```
public static int longestPath(Map<String, List<String>> graph) {
    HashMap<String, Integer> distance = new HashMap<>();
    for (String node : graph.keySet()) {
      if (graph.get(node).size() == 0) {
        distance.put(node, 0);
      }
    }
    
    for (String node : graph.keySet()) {
      traverseDistance(node, graph, distance);
    }
    
    return Collections.max(distance.values());
  }

public static int traverseDistance(String node, Map<String, List<String>> graph, HashMap<String, Integer> distance) {
    if (distance.containsKey(node)) {
      return distance.get(node);
    }
    
    int max = 0;
    for (String neighbor : graph.get(node)) {
      int attempt = traverseDistance(neighbor, graph, distance);
      if (attempt > max) {
        max = attempt;
      }
    }
    
    distance.put(node, max + 1);
    return max + 1;
  }
```

### semesters required
Write a method, semestersRequired, that takes in a number of courses (n) and a list of prerequisites as arguments. Courses have ids ranging from 0 through n - 1.  public static int semestersRequired(int numCourses, List<List<Integer>> prereqs) {
```
    HashMap<Integer, List<Integer>> graph = buildGraph(numCourses, prereqs);
    HashMap<Integer, Integer> distance = new HashMap<>();
    
    for (int node : graph.keySet()) {
      traverseDistance(node, graph, distance);
    }
    
    return Collections.max(distance.values());
  }

public static int traverseDistance(int node, HashMap<Integer, List<Integer>> graph, HashMap<Integer, Integer> distance) {
    if (distance.containsKey(node)) {
      return distance.get(node);
    }
    
    int max = 0;
    for (int neighbor : graph.get(node)) {
      int neighborDistance = traverseDistance(neighbor, graph, distance);
      if (neighborDistance > max) {
        max = neighborDistance;
      }
    }
    
    distance.put(node, max + 1);
    return max + 1;
  }

public static HashMap<Integer, List<Integer>> buildGraph(int numCourses, List<List<Integer>> prereqs) {
    HashMap<Integer, List<Integer>> graph = new HashMap<>();
    for (int node = 0; node < numCourses; node += 1) {
      graph.put(node, new ArrayList<>());
    }
    
    for (List<Integer> prereq : prereqs) {
      int a = prereq.get(0);
      int b = prereq.get(1);
      graph.get(a).add(b);
    }
    
    return graph;
  }
```
